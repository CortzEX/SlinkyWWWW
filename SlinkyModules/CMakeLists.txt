cmake_minimum_required(VERSION 3.20)
project(ShareableModules LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use Windows x64 with MSVC or clang-cl.")
endif()
file(GLOB MODULES CONFIGURE_DEPENDS "modules/*.cpp")
file(GLOB_RECURSE SUPPORT CONFIGURE_DEPENDS "runtime/shared/*.cpp" "runtime/native/verified/*.cpp")
add_library(shareable_module_objects OBJECT ${MODULES} ${SUPPORT})
target_compile_features(shareable_module_objects PUBLIC cxx_std_20)
target_compile_options(shareable_module_objects PRIVATE /EHsc /fp:strict /bigobj)
target_include_directories(shareable_module_objects PUBLIC "runtime/native/verified" "runtime/core")
add_library(shareable_modules STATIC $<TARGET_OBJECTS:shareable_module_objects>)
add_library(shareable_link_check SHARED EXCLUDE_FROM_ALL $<TARGET_OBJECTS:shareable_module_objects>)
